import {useEffect,useRef,useState,useSyncExternalStore} from 'react';
import {TaskController} from './tasks.ts';
export function TaskWorkspace({userId,projectId,onSessionLost,controller:provided}:{userId:string;projectId:string;onSessionLost:()=>void;controller?:TaskController}){
 const [controller]=useState(()=>provided??new TaskController(userId,projectId,onSessionLost));
 const state=useSyncExternalStore(controller.subscribe,controller.getSnapshot,controller.getSnapshot),feedback=useRef<HTMLParagraphElement>(null),previousBusy=useRef(false);
 useEffect(()=>{void controller.refresh();return()=>controller.cancel();},[controller]);
 useEffect(()=>{if(previousBusy.current&&!state.busy)feedback.current?.focus();previousBusy.current=state.busy;},[state.busy]);
 function create(event:React.SubmitEvent<HTMLFormElement>){event.preventDefault();const form=event.currentTarget,values=new FormData(form),title=String(values.get('title')??''),field=form.elements.namedItem('title') as HTMLInputElement;field.setCustomValidity(title.trim()?'':'Enter a task title.');if(!field.reportValidity())return;void controller.create(String(values.get('type')??''),title,String(values.get('description')??''));}
 const locked=state.busy||state.pending,detail=state.detail;
 return <section className="task-workspace" aria-labelledby="tasks-title" aria-busy={state.busy}>
  <h4 id="tasks-title">Tasks</h4><p role="status" aria-live="polite" tabIndex={-1} ref={feedback}>{state.message}</p>
  <div className="actions"><button disabled={state.busy} onClick={()=>void controller.refresh()}>Refresh tasks</button>{state.pending?<button className="primary" disabled={state.busy} onClick={()=>void controller.retry()}>Check original creation</button>:null}</div>
  <label htmlFor="task-location">Show tasks</label><select id="task-location" value={state.location} disabled={state.busy} onChange={e=>void controller.refresh(e.target.value as 'all'|'backlog')}><option value="all">All tasks</option><option value="backlog">Backlog</option></select>
  {!state.busy&&state.actions.includes('listTasks')&&!detail?<>
   {!state.items.length?<p>No tasks in this view.</p>:<ul className="task-list">{state.items.map(t=><li key={t.task_id}><button onClick={()=>void controller.open(t.task_id)}>{t.title}</button><p className="muted">{t.type} · {t.state.replaceAll('_',' ')} · {t.work_location.kind==='backlog'?'Backlog':t.work_location.kind==='completed'?'Completed Sprint':'Sprint'}{t.blocked?' · Blocked':''}</p></li>)}</ul>}
   {state.cursor?<button disabled={state.busy} onClick={()=>void controller.more()}>More tasks</button>:null}
  </>:null}
  {!state.busy&&detail?<article className="task-detail" aria-labelledby="task-detail-title"><h5 id="task-detail-title">{detail.title}</h5><p className="task-description">{detail.description||'No description.'}</p><dl><dt>Type</dt><dd>{detail.type}</dd><dt>State</dt><dd>{detail.state.replaceAll('_',' ')}</dd><dt>Location</dt><dd>{detail.work_location.kind==='backlog'?'Backlog · position '+detail.work_location.position:(detail.work_location.kind==='completed'?'Completed Sprint · ':'Sprint · ')+detail.work_location.sprint_id}</dd><dt>Created</dt><dd><time dateTime={detail.created_at}>{new Intl.DateTimeFormat(undefined,{dateStyle:'medium',timeStyle:'short'}).format(new Date(detail.created_at))}</time></dd><dt>Assignee</dt><dd>{detail.assignee_id??'Unassigned'}</dd><dt>Blocker status</dt><dd>{detail.blocked?'Blocked':'Not blocked'}</dd><dt>Revision</dt><dd>{detail.revision}</dd><dt>Task ID</dt><dd translate="no">{detail.task_id}</dd></dl><button onClick={()=>void controller.refresh()}>Back to tasks</button></article>:null}
  {!state.busy&&state.actions.includes('createTask')?<form onSubmit={create} autoComplete="off"><h5>Create a task</h5><label htmlFor="task-type">Task type</label><select id="task-type" name="type" defaultValue="task" disabled={locked}><option value="story">Story</option><option value="task">Task</option><option value="bug">Bug</option></select><label htmlFor="task-title">Task title</label><input id="task-title" name="title" required disabled={locked} onInput={e=>e.currentTarget.setCustomValidity('')}/><label htmlFor="task-description">Task description</label><textarea id="task-description" name="description" disabled={locked}/><button className="primary" disabled={locked}>Create task</button></form>:null}
 </section>;
}
